// CRUD de cursos document.addEventListener('DOMContentLoaded', function() { if (typeof firebase !== 'undefined' && firebase.firestore) { const db = firebase.firestore(); const cursosContent = document.getElementById('cursosContent'); const cursoForm = document.getElementById('cursoForm'); const cursoId = document.getElementById('cursoId'); const nombreCurso = document.getElementById('nombreCurso'); const docenteCurso = document.getElementById('docenteCurso'); const costoCurso = document.getElementById('costoCurso'); const temasCurso = document.getElementById('temasCurso'); const cursoFormMsg = document.getElementById('cursoFormMsg'); function cargarCursos() { db.collection('courses').get().then(snap => { let html = ''; snap.forEach(doc => { const c = doc.data(); html += ``; }); html += '
NombreDocenteTemasCosto (S/)
${c.nombre||'-'} ${c.docente||'-'} ${Array.isArray(c.temas)?c.temas.join(', '):(c.temas||'-')} S/ ${c.costo||'-'}
'; cursosContent.innerHTML = html; }).catch(()=>{ cursosContent.textContent = 'Error al cargar cursos'; }); } window.editarCurso = function(id) { db.collection('courses').doc(id).get().then(doc => { if (!doc.exists) return; const c = doc.data(); cursoId.value = id; nombreCurso.value = c.nombre||''; docenteCurso.value = c.docente||''; costoCurso.value = c.costo||''; temasCurso.value = Array.isArray(c.temas)?c.temas.join(', '):(c.temas||''); window.scrollTo({top:cursoForm.offsetTop-60,behavior:'smooth'}); }); } if (cursoForm) { cursoForm.onsubmit = function(e) { e.preventDefault(); const data = { nombre: nombreCurso.value.trim(), docente: docenteCurso.value.trim(), costo: parseFloat(costoCurso.value), temas: temasCurso.value.split(',').map(t=>t.trim()).filter(Boolean) }; cursoFormMsg.textContent = ''; let ref; if (cursoId.value) { ref = db.collection('courses').doc(cursoId.value).set(data, {merge:true}); } else { ref = db.collection('courses').add(data); } ref.then(()=>{ cursoForm.reset(); cursoId.value = ''; cursoFormMsg.textContent = 'Curso guardado correctamente'; cargarCursos(); setTimeout(()=>{cursoFormMsg.textContent='';},2000); }).catch(()=>{ cursoFormMsg.textContent = 'Error al guardar curso'; }); } } cargarCursos(); } });
